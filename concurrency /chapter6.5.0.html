One Million Concurrent HTTP Requests

try (var client = HttpClient.newBuilder()
         .executor(Executors.newVirtualThreadPerTaskExecutor())
         .build()) {

    List requests = IntStream.range(0, 1_000_000)
        .mapToObj(i -> HttpRequest.newBuilder()
            .uri(URI.create("https://httpbin.org/delay/1"))
            .build())
        .toList();

    List> futures = requests.stream()
        .map(req -> client.sendAsync(req, BodyHandlers.ofString())
            .thenApply(HttpResponse::body))
        .toList();

    CompletableFuture.allOf(futures.toArray(CompletableFuture[]::new))
        .join();

    System.out.println("All 1,000,000 requests completed!");
}

• ~1.2 GB RAM used
