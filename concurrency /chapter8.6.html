Java-Only Solution: Seat Locking with ReentrantLock

public class SeatLocker {
    private final Map locks = new ConcurrentHashMap<>();
    private final ScheduledExecutorService cleaner = Executors.newScheduledThreadPool(1);

    public boolean lockSeat(String seatId, String userId, long timeoutSeconds) {
        ReentrantLock lock = locks.computeIfAbsent(seatId, k -> new ReentrantLock());

        boolean acquired = false;
        try {
            acquired = lock.tryLock(timeoutSeconds, TimeUnit.SECONDS);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        if (acquired) {
            // Schedule unlock
            cleaner.schedule(() -> {
                lock.unlock();
                locks.remove(seatId, lock);
            }, timeoutSeconds, TimeUnit.SECONDS);
        }
        return acquired;
    }
}
