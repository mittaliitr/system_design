Complete Working Simulator (Java 21+)

import java.time.*;
import java.util.*;
import java.util.concurrent.*;

public class MovieBookingSimulator {
    private static final int TOTAL_SEATS = 100;
    private static final int TOTAL_USERS = 1000;
    private static final Duration LOCK_TIMEOUT = Duration.ofMinutes(5);

    record LockInfo(String userId, Instant expiresAt) {}

    private final Map seatLocks = new ConcurrentHashMap<>();
    private final Set bookedSeats = ConcurrentHashMap.newKeySet();
    private final Random random = new Random();

    public static void main(String[] args) throws Exception {
        new MovieBookingSimulator().run();
    }

    void run() throws Exception {
        System.out.println("Avengers: Endgame — " + TOTAL_SEATS + " seats");
        System.out.println(TOTAL_USERS + " users attacking...");

        try (var scope = new StructuredTaskScope.ShutdownOnFailure();
             var executor = Executors.newVirtualThreadPerTaskExecutor()) {

            List> futures = new ArrayList<>();
            for (int i = 0; i < TOTAL_USERS; i++) {
                int userId = i;
                futures.add(scope.fork(() -> simulateUser("User-" + userId)));
            }

            scope.join();           // Wait for all
            scope.throwIfFailed();
        }

        printFinalResults();
    }

    void simulateUser(String userId) {
        List desiredSeats = selectRandomSeats(1 + random.nextInt(4));

        if (tryLockSeats(userId, desiredSeats)) {
            // Simulate payment thinking time
            sleep(random.nextInt(3000, 10000));

            if (isLockValid(userId, desiredSeats)) {
                bookedSeats.addAll(desiredSeats);
                System.out.println(userId + " BOOKED " + desiredSeats);
            }
            releaseSeats(userId, desiredSeats);
        } else {
            System.out.println(userId + " failed to get " + desiredSeats);
        }
    }

    private boolean tryLockSeats(String userId, List seats) {
        Instant expires = Instant.now().plus(LOCK_TIMEOUT);

        for (String seat : seats) {
            LockInfo existing = seatLocks.putIfAbsent(seat, new LockInfo(userId, expires));
            if (existing != null && existing.expiresAt().isAfter(Instant.now())) {
                // Clean expired but not removed
                seatLocks.entrySet().removeIf(e -> e.getValue().expiresAt().isBefore(Instant.now()));
                return false; // retry logic in real system
            }
        }
        // Success — update with same expiry
        seats.forEach(seat -> seatLocks.put(seat, new LockInfo(userId, expires)));
        return true;
    }

    private void releaseSeats(String userId, List seats) {
        seats.forEach(seatLocks::remove);
    }

    private boolean isLockValid(String userId, List seats) {
        return seats.stream()
            .allMatch(seat -> {
                LockInfo lock = seatLocks.get(seat);
                return lock != null && lock.userId().equals(userId) &&
                       lock.expiresAt().isAfter(Instant.now());
            });
    }

    private List selectRandomSeats(int count) {
        List seats = new ArrayList<>();
        for (int i = 0; i < count; i++) {
            seats.add("A" + (1 + random.nextInt(TOTAL_SEATS)));
        }
        return seats;
    }

    private void sleep(long ms) {
        try { Thread.sleep(ms); } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
    }

    private void printFinalResults() {
        System.out.println("\n=== FINAL RESULTS ===");
        System.out.println("Booked seats: " + bookedSeats.size());
        System.out.println("Available seats: " + (TOTAL_SEATS - bookedSeats.size()));
        System.out.println("No double booking occurred!");
    }
}
