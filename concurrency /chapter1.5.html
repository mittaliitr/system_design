Case Study: Spring Boot Microservice

E-Commerce Order Processing Service

A single JVM instance handles 50,000 orders/min using 200 threads. 

@RestController
class OrderController {
    private final OrderService orderService;

    @PostMapping("/orders")
    public ResponseEntity create(@RequestBody OrderRequest req) {
        // Non-blocking: offload to thread pool
        return orderService.processAsync(req);
    }
}

@Service
class OrderService {
    private final TaskExecutor executor;

    public ResponseEntity processAsync(OrderRequest req) {
        CompletableFuture.supplyAsync(() -> process(req), executor)
            .thenApply(ResponseEntity::ok);
    }
}

// Main (UI) Thread
button.setOnClickListener(v -> {
    // Never block UI thread!

        Bitmap image = downloadImage(url);
        runOnUiThread(() -> imageView.setImageBitmap(image));
    }).start();
