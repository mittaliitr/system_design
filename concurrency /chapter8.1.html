Core Implementation (Java 21+)

public class ParallelDownloader {
    private static final int CHUNK_SIZE = 5 * 1024 * 1024; // 5MB
    private final URI uri;
    private final Path output;
    private final int parallelism;

    public static void main(String[] args) throws Exception {
        new ParallelDownloader(URI.create(args[0]), Path.of("download.file"))
            .download();
    }

    public void download() throws Exception {
        long fileSize = getFileSize();
        long chunks = (fileSize + CHUNK_SIZE - 1) / CHUNK_SIZE;

        try (var scope = new StructuredTaskScope.ShutdownOnFailure()) {
            List> futures = new ArrayList<>();
            for (int i = 0; i < chunks; i++) {
                long start = i * CHUNK_SIZE;
                long end = Math.min(start + CHUNK_SIZE, fileSize) - 1;
                int chunk = i;
                futures.add(scope.fork(() -> downloadChunk(chunk, start, end)));
            }

            scope.join();           // Wait for all
            scope.throwIfFailed();  // Propagate error
        }
    }

    private Void downloadChunk(int chunk, long start, long end) throws Exception {
        HttpRequest request = HttpRequest.newBuilder()
            .uri(uri)
            .header("Range", "bytes=" + start + "-" + end)
            .build();

        byte[] data = client.send(request, BodyHandlers.ofByteArray()).body();
        Files.write(output.resolveSibling(output.getFileName() + ".part" + chunk),
            RandomAccessFile raf = new RandomAccessFile(output.toFile(), "rw");
        raf.seek(start);
        raf.write(data);
        raf.close();

        updateProgress();
        return null;
    }
}
Key Points:
• StructuredTaskScope → automatic cancellation
• HttpClient with Range headers
• Write directly to correct offset
• Progress bar with atomic counter
