ForkBlur.java — The Classic Example

public class ForkBlur extends RecursiveAction {
    private final int[] source;
    private final int start;
    private final int length;
    private final int[] destination;
    private final int blurWidth = 15;

    public ForkBlur(int[] src, int start, int length, int[] dst) {
        this.source = src;
        this.start = start;
        this.length = length;
        this.destination = dst;
    }

    protected void computeDirectly() {
        for (int i = start; i < start + length; i++) {
            float r = 0, g = 0, b = 0;
            for (int j = -blurWidth; j <= blurWidth; j++) {
                int idx = Math.min(Math.max(i + j, 0), source.length - 1);
                r += (source[idx] >> 16) & 0xFF;
                g += (source[idx] >> 8) & 0xFF;
                b += source[idx] & 0xFF;
            }
            int divisor = blurWidth * 2 + 1;
            int rgb = ((int)(r/divisor) << 16) | ((int)(g/divisor) << 8) | (int)(b/divisor);
            destination[i] = rgb;
        }
    }

    private static final int THRESHOLD = 100_000;

    @Override
    protected void compute() {
        if (length < THRESHOLD) {
            computeDirectly();
            return;
        }

        int split = length / 2;

        ForkBlur left = new ForkBlur(source, start, split, destination);
        ForkBlur right = new ForkBlur(source, start + split, length - split, destination);

        left.fork();
        right.compute();
        left.join();
    }

    // Main method to run
    public static void main(String[] args) throws Exception {
        BufferedImage img = ImageIO.read(new File("input.jpg"));
        int[] src = getRGB(img);
        int[] dst = new int[src.length];

        ForkJoinPool pool = new ForkJoinPool();
        pool.invoke(new ForkBlur(src, 0, src.length, dst));

        BufferedImage blurred = createImage(dst, img.getWidth(), img.getHeight());
        ImageIO.write(blurred, "png", new File("blurred.png"));
    }
}
