LeetCode 825: Friends of Appropriate Ages

Problem Explanation

You are given an integer array ages where ages[i] represents the age of the i-th person. A person x will not send a friend request to a person y (where x != y) if any of the following are true:

  • age[y] <= 0.5 * age[x] + 7
  • age[y] > age[x]
  • age[y] > 100 && age[x] < 100

Your task is to return the total number of friend requests that can be made.

Approach

  1. Sort the ages array.
  2. For each person's age ageX, calculate the lower and upper bounds for valid friend requests:
    3. lower = 0.5 * ageX + 7
upper = ageX
  3. Use binary search to find how many people fall in the valid range (lower, upper].
  4. Count all valid friend requests.

Java Code Implementation

import java.util.Arrays;

public class Solution {
    public int numFriendRequests(int[] ages) {
        Arrays.sort(ages);
        int n = ages.length;
        int count = 0;

        for (int i = 0; i < n; i++) {
            int ageX = ages[i];
            int lower = (int) (0.5 * ageX + 7);
            int upper = ageX;

            int left = binarySearchLower(ages, lower + 1);
            int right = binarySearchUpper(ages, upper);

            count += (right - left);
        }

        return count;
    }

    private int binarySearchLower(int[] ages, int target) {
        int left = 0, right = ages.length - 1;
        int ans = ages.length;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (ages[mid] >= target) {
                ans = mid;
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        }
        return ans;
    }

    private int binarySearchUpper(int[] ages, int target) {
        int left = 0, right = ages.length - 1;
        int ans = -1;
        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (ages[mid] <= target) {
                ans = mid;
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        return ans + 1;
    }

    public static void main(String[] args) {
        Solution sol = new Solution();
        int[] ages = {16, 16};
        System.out.println(sol.numFriendRequests(ages)); // Output: 2
    }
}
