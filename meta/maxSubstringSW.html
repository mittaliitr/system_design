76. Minimum Window Substring — Java + JavaScript Explained

Problem: Given two strings s and t, return the smallest substring of s that contains all characters of t. If no such substring exists, return "".

Example

Input: s = "ADOBECODEBANC", t = "ABC"
Output: "BANC"

Approach (Sliding Window)

  1. Use two pointers (left and right) to track a moving window.
  2. Expand the right pointer until all required characters from t are included.
  3. Then, move the left pointer to shrink the window while keeping it valid.
  4. Keep track of the smallest valid window found so far.

✅ JavaScript Implementation


function minWindow(s, t) {
  if (t.length === 0 || s.length === 0) return "";

  const need = new Map();
  for (let ch of t) {
    need.set(ch, (need.get(ch) || 0) + 1);
  }

  let have = 0;
  const windowCountsObj = new Map();
  const required = need.size;

  let res = [-1, -1];
  let resLen = Infinity;
  let l = 0;

  for (let r = 0; r < s.length; r++) {
    const ch = s[r];
    windowCountsObj.set(ch, (windowCountsObj.get(ch) || 0) + 1);

    if (need.has(ch) && windowCountsObj.get(ch) === need.get(ch)) {
      have++;
    }

    while (have === required) {
      if ((r - l + 1) < resLen) {
        res = [l, r];
        resLen = r - l + 1;
      }

      const leftChar = s[l];
      windowCountsObj.set(leftChar, windowCountsObj.get(leftChar) - 1);
      if (need.has(leftChar) && windowCountsObj.get(leftChar) < need.get(leftChar)) {
        have--;
      }
      l++;
    }
  }

  const [start, end] = res;
  return resLen === Infinity ? "" : s.slice(start, end + 1);
}

// Test
console.log(minWindow("ADOBECODEBANC", "ABC")); // BANC

🧠 Java Implementation


import java.util.*;

public class MinimumWindowSubstring {
    public static String minWindow(String s, String t) {
        if (s.length() == 0 || t.length() == 0) return "";

        Map<Character, Integer> need = new HashMap<>();
        for (char c : t.toCharArray()) {
            need.put(c, need.getOrDefault(c, 0) + 1);
        }

        Map<Character, Integer> windowCounts = new HashMap<>();
        int have = 0, required = need.size();
        int left = 0, minLen = Integer.MAX_VALUE;
        int start = 0;

        for (int right = 0; right < s.length(); right++) {
            char c = s.charAt(right);
            windowCounts.put(c, windowCounts.getOrDefault(c, 0) + 1);

            if (need.containsKey(c) && windowCounts.get(c).intValue() == need.get(c).intValue()) {
                have++;
            }

            while (have == required) {
                if ((right - left + 1) < minLen) {
                    start = left;
                    minLen = right - left + 1;
                }

                char leftChar = s.charAt(left);
                windowCounts.put(leftChar, windowCounts.get(leftChar) - 1);
                if (need.containsKey(leftChar) && windowCounts.get(leftChar) < need.get(leftChar)) {
                    have--;
                }
                left++;
            }
        }

        return minLen == Integer.MAX_VALUE ? "" : s.substring(start, start + minLen);
    }

    public static void main(String[] args) {
        System.out.println(minWindow("ADOBECODEBANC", "ABC")); // BANC
        System.out.println(minWindow("a", "a")); // a
        System.out.println(minWindow("a", "aa")); // ""
    }
}
