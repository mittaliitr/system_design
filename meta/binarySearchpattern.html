Binary Search Boundary Demo

🧩 Java Code:


public class BinarySearchBoundary {

    public static void main(String[] args) {
        boolean[] nums = new boolean[]{false, false, true, true, true};
        int result = binarySearch(nums);
        System.out.println("First true index: " + result);
    }

    public static int binarySearch(boolean[] nums) {
        int left = 0;
        int right = nums.length - 1;
        int ans = -1;

        while (left <= right) {
            int mid = left + (right - left) / 2;
            if (nums[mid]) {
                ans = mid;
                right = mid - 1;
            } else {
                left = mid + 1;
            }
        }

        return ans;
    }
}
